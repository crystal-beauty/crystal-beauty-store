     1→"use client";
     2→
     3→import { useMemo, useState, useEffect } from "react";
     4→import { useRouter } from "next/navigation";
     5→import { ShoppingBag, Gift, Package, Banknote, Truck, PackageSearch, Check, Minus, Plus, Loader2, Star, MessageCircle } from "lucide-react";
     6→import { useCart } from "@/lib/cart";
     7→import { trackPixel } from "@/lib/pixel";
     8→import { useSettings } from "@/components/store/settings-provider";
     9→import { formatMoney, isValidDzMobile, toInternationalDz } from "@/lib/format";
    10→import type { MediaItem } from "@/lib/format";
    11→import { useT } from "@/components/store/locale-provider";
    12→import DeliverySelector, { emptyDelivery, type DeliveryValue } from "@/components/store/delivery-selector";
    13→import { cn } from "@/lib/utils";
    14→import { toast } from "sonner";
    15→import { Accordion, AccordionContent, AccordionItem, AccordionTrigger } from "@/components/ui/accordion";
    16→
    17→interface VariantData {
    18→  id: string;
    19→  name: string;
    20→  required: boolean;
    21→  options: { id: string; label: string; priceModifier: number; hexColor: string | null; stock: number | null }[];
    22→}
    23→
    24→export interface ProductReview {
    25→  id: string;
    26→  customerName: string;
    27→  wilayaName: string;
    28→  rating: number;
    29→  text: string;
    30→  createdAt: string;
    31→}
    32→
    33→interface ProductData {
    34→  id: string;
    35→  name: string;
    36→  slug: string;
    37→  sku: string | null;
    38→  price: number;
    39→  comparePrice: number | null;
    40→  stock: number;
    41→  trackStock: boolean;
    42→  shortDescription: string | null;
    43→  description: string | null;
    44→  media: MediaItem[];
    45→  variants: VariantData[];
    46→}
    47→
    48→export default function ProductDetail({
    49→  product,
    50→  giftBox,
    51→  reviews,
    52→  whatsappNumber,
    53→}: {
    54→  product: ProductData;
    55→  giftBox: { enabled: boolean; fee: number; label: string; labelAr: string; description: string; descriptionAr: string };
    56→  reviews: ProductReview[];
    57→  whatsappNumber: string;
    58→}) {
    59→  const router = useRouter();
    60→  const settings = useSettings();
    61→  const t = useT();
    62→  const addItem = useCart((s) => s.addItem);
    63→
    64→  const [activeMedia, setActiveMedia] = useState(0);
    65→  const [quantity, setQuantity] = useState(1);
    66→  const [wantGiftBox, setWantGiftBox] = useState(false);
    67→  const [selections, setSelections] = useState<Record<string, string>>(() => {
    68→    const initial: Record<string, string> = {};
    69→    for (const v of product.variants) if (v.required && v.options.length > 0) initial[v.name] = v.options[0].label;
    70→    return initial;
    71→  });
    72→
    73→  // ---- quick order form ----
    74→  const [form, setForm] = useState({ customerName: "", phone: "", address: "", notes: "" });
    75→  const [delivery, setDelivery] = useState<DeliveryValue>(emptyDelivery);
    76→  const [fee, setFee] = useState<number | null>(null);
    77→  const [feeState, setFeeState] = useState<"idle" | "loading" | "error">("idle");
    78→  const [submitting, setSubmitting] = useState(false);
    79→
    80→  // ---- pricing ----
    81→  const unitPrice = useMemo(() => {
    82→    let price = product.price;
    83→    for (const v of product.variants) {
    84→      const chosen = selections[v.name];
    85→      const opt = v.options.find((o) => o.label === chosen);
    86→      if (opt) price += opt.priceModifier;
    87→    }
    88→    return Math.round(price);
    89→  }, [product.price, product.variants, selections]);
    90→
    91→  const maxStock = product.trackStock ? product.stock : 0;
    92→  const soldOut = product.trackStock && product.stock <= 0;
    93→  const giftBoxActive = giftBox.enabled && wantGiftBox;
    94→  const lineTotal = unitPrice * quantity + (giftBoxActive ? giftBox.fee * quantity : 0);
    95→
    96→  // ---- Meta Pixel: ViewContent ----
    97→  useEffect(() => {
    98→    trackPixel("ViewContent", {
    99→      customData: {
   100→        content_ids: [product.id],
   101→        content_name: product.name,
   102→        content_type: "product",
   103→        value: product.price,
   104→        currency: "DZD",
   105→      },
   106→    });
   107→  }, [product.id]);
   108→
   109→  function buildItem() {
   110→    return {
   111→      productId: product.id,
   112→      slug: product.slug,
   113→      name: product.name,
   114→      image: product.media[0]?.url || "",
   115→      unitPrice,
   116→      basePrice: product.price,
   117→      quantity,
   118→      selections: { ...selections },
   119→      giftBox: giftBoxActive,
   120→      maxStock,
   121→    };
   122→  }
   123→
   124→  function handleAddToCart() {
   125→    if (soldOut) return;
   126→    addItem(buildItem());
   127→    trackPixel("AddToCart", {
   128→      customData: {
   129→        content_ids: [product.id],
   130→        content_name: product.name,
   131→        content_type: "product",
   132→        value: unitPrice * quantity,
   133→        currency: "DZD",
   134→        contents: [{ id: product.id, quantity, item_price: unitPrice }],
   135→      },
   136→    });
   137→    toast.success(`${product.name} ✓`);
   138→  }
   139→
   140→  function handleBuyNow() {
   141→    if (soldOut) return;
   142→    addItem(buildItem());
   143→    trackPixel("AddToCart", {
   144→      customData: {
   145→        content_ids: [product.id],
   146→        content_name: product.name,
   147→        content_type: "product",
   148→        value: unitPrice * quantity,
   149→        currency: "DZD",
   150→        contents: [{ id: product.id, quantity, item_price: unitPrice }],
   151→      },
   152→    });
   153→    trackPixel("InitiateCheckout", {
   154→      customData: { value: lineTotal, currency: "DZD", num_items: quantity, content_ids: [product.id] },
   155→    });
   156→    router.push("/checkout");
   157→  }
   158→
   159→  function selectGiftBox(checked: boolean) {
   160→    setWantGiftBox(checked);
   161→    if (checked)
   162→      trackPixel("GiftBoxSelected", { customData: { content_name: product.name, value: giftBox.fee, currency: "DZD" } });
   163→  }
   164→
   165→  // ---- quick order submit (COD) ----
   166→  async function handleQuickOrder(e: React.FormEvent) {
   167→    e.preventDefault();
   168→    if (soldOut) return;
   169→    if (!isValidDzMobile(form.phone)) {
   170→      toast.error(t("phoneInvalid"));
   171→      return;
   172→    }
   173→    if (!delivery.wilayaId || !delivery.commune) {
   174→      toast.error(t("selectBoth"));
   175→      return;
   176→    }
   177→    if (fee === null || feeState !== "idle") {
   178→      toast.error(t("feeError"));
   179→      return;
   180→    }
   181→    setSubmitting(true);
   182→    try {
   183→      const res = await fetch("/api/orders", {
   184→        method: "POST",
   185→        headers: { "Content-Type": "application/json" },
   186→        body: JSON.stringify({
   187→          ...form,
   188→          wilayaId: delivery.wilayaId,
   189→          commune: delivery.commune,
   190→          deliveryType: delivery.deliveryType,
   191→          items: [
   192→            {
   193→              productId: product.id,
   194→              quantity,
   195→              selections,
   196→              giftBox: giftBoxActive,
   197→            },
   198→          ],
   199→        }),
   200→      });
   201→      const data = await res.json();
   202→      if (!res.ok) {
   203→        toast.error(data.error || t("errorGeneric"));
   204→        setSubmitting(false);
   205→        return;
   206→      }
   207→      trackPixel("AddPaymentInfo", { customData: { value: lineTotal + (fee ?? 0), currency: "DZD", payment_method: "COD" } });
   208→      router.push(`/order-confirmation/${data.orderNumber}?event=${encodeURIComponent(data.pixelEventId)}`);
   209→    } catch {
   210→      toast.error(t("errorGeneric"));
   211→      setSubmitting(false);
   212→    }
   213→  }
   214→
   215→  const media = product.media;
   216→  const inputCls =
   217→    "w-full bg-white border border-input rounded-xl px-4 py-3 text-sm focus:outline-none focus:ring-2 focus:ring-[var(--primary)]/25 focus:border-[var(--primary)] transition-all";
   218→  const labelCls = "block text-sm font-medium text-[var(--primary)] mb-1.5";
   219→  const waHref = whatsappNumber
   220→    ? `https://wa.me/${toInternationalDz(whatsappNumber).replace("+", "")}?text=${encodeURIComponent(`Bonjour / مرحبًا — ${product.name}`)}`
   221→    : "";
   222→
   223→  return (
   224→    <div className="pb-24 lg:pb-0">
   225→      <div className="grid lg:grid-cols-2 gap-10 lg:gap-16">
   226→        {/* ============ GALLERY — sticky on desktop ============ */}
   227→        <div className="lg:sticky lg:top-28 lg:self-start">
   228→          <div className="relative overflow-hidden rounded-2xl bg-muted aspect-square shadow-luxe">
   229→            {media[activeMedia]?.type === "video" ? (
   230→              <video key={media[activeMedia].url} src={media[activeMedia].url} controls playsInline className="h-full w-full object-cover" />
   231→            ) : media[activeMedia] ? (
   232→              <img key={media[activeMedia].url} src={media[activeMedia].url} alt={media[activeMedia].alt || product.name} className="h-full w-full object-cover" />
   233→            ) : (
   234→              <div className="h-full w-full flex items-center justify-center">
   235→                <Package className="h-16 w-16 text-muted-foreground/30" />
   236→              </div>
   237→            )}
   238→          </div>
   239→
   240→          {media.length > 1 && (
   241→            <div className="flex gap-3 mt-4 overflow-x-auto no-scrollbar pb-1">
   242→              {media.map((m, i) => (
   243→                <button
   244→                  key={i}
   245→                  onClick={() => setActiveMedia(i)}
   246→                  aria-label={`Media ${i + 1}`}
   247→                  className={cn(
   248→                    "relative shrink-0 h-20 w-20 rounded-xl overflow-hidden border-2 transition-all",
   249→                    activeMedia === i ? "border-[var(--primary)]" : "border-transparent opacity-70 hover:opacity-100"
   250→                  )}
   251→                >
   252→                  {m.type === "video" ? (
   253→                    <video src={m.url} className="h-full w-full object-cover" muted />
   254→                  ) : (
   255→                    <img src={m.url} alt={m.alt || product.name} className="h-full w-full object-cover" loading="lazy" />
   256→                  )}
   257→                  {m.type === "video" && (
   258→                    <span className="absolute inset-0 flex items-center justify-center bg-black/30 text-white text-xs font-bold">▶</span>
   259→                  )}
   260→                </button>
   261→              ))}
   262→            </div>
   263→          )}
   264→        </div>
   265→
   266→        {/* ============ DETAILS — clear groups with generous proximity ============ */}
   267→        <div className="flex flex-col gap-9">
   268→          {/* — Identity & price — */}
   269→          <section>
   270→            <h1 className="font-display text-3xl sm:text-4xl font-semibold text-[var(--primary)] leading-tight">{product.name}</h1>
   271→
   272→            {product.shortDescription && <p className="mt-3 text-muted-foreground leading-relaxed">{product.shortDescription}</p>}
   273→
   274→            <div className="mt-5 flex items-baseline gap-3 flex-wrap">
   275→              <span className="text-3xl font-semibold tracking-tight text-[var(--primary)]">{formatMoney(unitPrice)}</span>
   276→              {product.comparePrice && product.comparePrice > product.price && (
   277→                <>
   278→                  <span className="text-lg text-muted-foreground line-through">{formatMoney(product.comparePrice)}</span>
   279→                  <span className="self-center rounded-full bg-[var(--primary)]/5 border border-[var(--primary)]/10 px-2.5 py-0.5 text-xs font-semibold text-[var(--primary)]">
   280→                    -{Math.round((1 - product.price / product.comparePrice) * 100)}%
   281→                  </span>
   282→                </>
   283→              )}
   284→            </div>
   285→
   286→            {/* Real stock from the dashboard */}
   287→            {product.trackStock && (
   288→              <p className={cn("mt-3 text-sm font-medium", soldOut ? "text-destructive" : "text-[var(--primary)]")}>
   289→                {soldOut ? t("soldOut") : product.stock <= 5 ? `${t("lowStock")} ${product.stock} ${t("unitsLeft")}` : `${t("inStock")} (${product.stock})`}
   290→              </p>
   291→            )}
   292→          </section>
   293→
   294→          {/* — Options: variants + gift box (one tight group) — */}
   295→          {(product.variants.length > 0 || giftBox.enabled) && (
   296→            <section className="space-y-5">
   297→              {product.variants.map((variant) => (
   298→                <div key={variant.id}>
   299→                  <div className="flex items-center justify-between mb-2.5">
   300→                    <p className="text-sm font-semibold text-[var(--primary)]">
   301→                      {variant.name}
   302→                      {variant.required && <span className="text-destructive"> *</span>}
   303→                    </p>
   304→                    {selections[variant.name] && <p className="text-sm text-muted-foreground">{selections[variant.name]}</p>}
   305→                  </div>
   306→                  <div className="flex flex-wrap gap-2">
   307→                    {variant.options.map((option) => {
   308→                      const active = selections[variant.name] === option.label;
   309→                      return (
   310→                        <button
   311→                          key={option.id}
   312→                          onClick={() => setSelections((s) => ({ ...s, [variant.name]: option.label }))}
   313→                          aria-pressed={active}
   314→                          className={cn(
   315→                            "relative px-4 py-2.5 rounded-xl border text-sm font-medium transition-all flex items-center gap-2",
   316→                            active ? "border-[var(--primary)] bg-[var(--primary)] text-white" : "border-input bg-white text-[var(--primary)] hover:border-[var(--primary)]"
   317→                          )}
   318→                        >
   319→                          {option.hexColor && <span className="h-4 w-4 rounded-full border border-black/10 inline-block" style={{ backgroundColor: option.hexColor }} />}
   320→                          {option.label}
   321→                          {option.priceModifier !== 0 && (
   322→                            <span className={cn("text-xs", active ? "text-white/80" : "text-muted-foreground")}>
   323→                              {option.priceModifier > 0 ? `+${formatMoney(option.priceModifier)}` : `-${formatMoney(Math.abs(option.priceModifier))}`}
   324→                            </span>
   325→                          )}
   326→                        </button>
   327→                      );
   328→                    })}
   329→                  </div>
   330→                </div>
   331→              ))}
   332→
   333→              {/* Gift box add-on (optional — gifts are single-piece by default) */}
   334→              {giftBox.enabled && (
   335→                <label
   336→                  className={cn(
   337→                    "flex items-start gap-3.5 rounded-2xl border p-4 cursor-pointer transition-all",
   338→                    giftBoxActive ? "border-[var(--brand)] bg-[var(--brand)]/5" : "border-input bg-white hover:border-[var(--primary)]/40"
   339→                  )}
   340→                >
   341→                  <input type="checkbox" checked={giftBoxActive} onChange={(e) => selectGiftBox(e.target.checked)} className="mt-1 h-4 w-4 accent-[var(--brand)]" />
   342→                  <div className="flex-1">
   343→                    <div className="flex items-center gap-2 flex-wrap">
   344→                      <Gift className="h-4 w-4 text-[var(--brand)]" />
   345→                      <span className="text-sm font-semibold text-[var(--primary)]">{giftBox.label}</span>
   346→                      <span className="text-sm font-semibold text-[var(--brand)]">+{formatMoney(giftBox.fee)}</span>
   347→                    </div>
   348→                    <p className="text-xs text-muted-foreground mt-1 leading-relaxed">{giftBox.description}</p>
   349→                  </div>
   350→                </label>
   351→              )}
   352→            </section>
   353→          )}
   354→
   355→          {/* — Action: quantity, total, one clear primary CTA — */}
   356→          <section>
   357→            <div className="flex items-center gap-4">
   358→              <div className="flex items-center border border-input rounded-xl bg-white">
   359→                <button onClick={() => setQuantity((q) => Math.max(1, q - 1))} className="p-3 text-[var(--primary)] hover:bg-muted rounded-s-xl" aria-label="-">
   360→                  <Minus className="h-4 w-4" />
   361→                </button>
   362→                <span className="w-10 text-center font-semibold text-[var(--primary)]" aria-live="polite">
   363→                  {quantity}
   364→                </span>
   365→                <button
   366→                  onClick={() => setQuantity((q) => (maxStock > 0 ? Math.min(maxStock, q + 1) : Math.min(99, q + 1)))}
   367→                  className="p-3 text-[var(--primary)] hover:bg-muted rounded-e-xl"
   368→                  aria-label="+"
   369→                >
   370→                  <Plus className="h-4 w-4" />
   371→                </button>
   372→              </div>
   373→              <p className="text-sm text-muted-foreground">
   374→                {t("total")}: <span className="font-semibold text-[var(--primary)]">{formatMoney(lineTotal)}</span>
   375→              </p>
   376→            </div>
   377→
   378→            <div className="mt-5 grid gap-3 sm:grid-cols-2">
   379→              {/* Primary — the COD-native action */}
   380→              <button
   381→                onClick={handleBuyNow}
   382→                disabled={soldOut}
   383→                className="press inline-flex items-center justify-center gap-2 bg-[var(--primary)] text-white font-semibold py-4 rounded-full hover:bg-[#16305c] disabled:opacity-40 disabled:cursor-not-allowed"
   384→              >
   385→                <Banknote className="h-4.5 w-4.5" />
   386→                {t("orderNow")}
   387→              </button>
   388→              {/* Secondary — outline, clearly subordinate */}
   389→              <button
   390→                onClick={handleAddToCart}
   391→                disabled={soldOut}
   392→                className="press inline-flex items-center justify-center gap-2 border border-[var(--primary)]/25 bg-white text-[var(--primary)] font-semibold py-4 rounded-full hover:border-[var(--primary)] disabled:opacity-40 disabled:cursor-not-allowed"
   393→              >
   394→                <ShoppingBag className="h-4.5 w-4.5" />
   395→                {t("addToCart")}
   396→              </button>
   397→            </div>
   398→          </section>
   399→
   400→          {/* — Reassurance: flat honest list, real facts only — */}
   401→          <ul className="space-y-3 border-t border-border pt-7">
   402→            {[
   403→              { icon: Banknote, title: t("codBadge"), text: t("codBadgeText") },
   404→              { icon: Truck, title: t("wilayasBadge"), text: t("wilayasBadgeText") },
   405→              { icon: PackageSearch, title: t("trackBadge"), text: t("trackBadgeText") },
   406→            ].map(({ icon: Icon, title, text }) => (
   407→              <li key={title} className="flex items-start gap-3">
   408→                <Icon className="h-4.5 w-4.5 text-[var(--brand)] mt-0.5 shrink-0" />
   409→                <div>
   410→                  <p className="text-sm font-semibold text-[var(--primary)] leading-snug">{title}</p>
   411→                  <p className="text-xs text-muted-foreground leading-relaxed mt-0.5">{text}</p>
   412→                </div>
   413→              </li>
   414→            ))}
   415→          </ul>
   416→
   417→          {/* — Details — */}
   418→          <section>
   419→            <Accordion type="multiple" defaultValue={["description"]}>
   420→              {product.description && (
   421→                <AccordionItem value="description">
   422→                  <AccordionTrigger className="text-[var(--primary)] font-semibold">{t("description")}</AccordionTrigger>
   423→                  <AccordionContent>
   424→                    <div className="text-muted-foreground leading-relaxed whitespace-pre-line">{product.description}</div>
   425→                  </AccordionContent>
   426→                </AccordionItem>
   427→              )}
   428→              <AccordionItem value="delivery">
   429→                <AccordionTrigger className="text-[var(--primary)] font-semibold">{t("deliveryFee")}</AccordionTrigger>
   430→                <AccordionContent>
   431→                  <ul className="space-y-2 text-muted-foreground">
   432→                    <li className="flex items-start gap-2">
   433→                      <Check className="h-4 w-4 text-[var(--brand)] mt-0.5 shrink-0" />
   434→                      {t("deliveryInfo")}
   435→                    </li>
   436→                    <li className="flex items-start gap-2">
   437→                      <Check className="h-4 w-4 text-[var(--brand)] mt-0.5 shrink-0" />
   438→                      {t("codOnlyHint")}
   439→                    </li>
   440→                    <li className="flex items-start gap-2">
   441→                      <Check className="h-4 w-4 text-[var(--brand)] mt-0.5 shrink-0" />
   442→                      {t("giftBadgeText")}
   443→                    </li>
   444→                  </ul>
   445→                </AccordionContent>
   446→              </AccordionItem>
   447→            </Accordion>
   448→
   449→            {product.sku && (
   450→              <p className="mt-4 text-xs text-muted-foreground" dir="ltr">
   451→                SKU: {product.sku}
   452→              </p>
   453→            )}
   454→          </section>
   455→        </div>
   456→      </div>
   457→
   458→      {/* ============ QUICK ORDER (COD) ============ */}
   459→      <section id="quick-order" className="mt-16 scroll-mt-24">
   460→        <div className="rounded-3xl border border-border bg-white shadow-luxe p-6 sm:p-10 max-w-3xl mx-auto">
   461→          <h2 className="font-display text-2xl sm:text-3xl font-semibold text-[var(--primary)]">{t("quickOrder")}</h2>
   462→          <p className="mt-2 text-sm text-muted-foreground">{t("quickOrderSubtitle")}</p>
   463→
   464→          <form onSubmit={handleQuickOrder} className="mt-8 space-y-5">
   465→            <div className="grid sm:grid-cols-2 gap-4">
   466→              <div>
   467→                <label className={labelCls} htmlFor="qo-name">
   468→                  {t("yourName")} *
   469→                </label>
   470→                <input id="qo-name" required value={form.customerName} onChange={(e) => setForm((f) => ({ ...f, customerName: e.target.value }))} className={inputCls} />
   471→              </div>
   472→              <div>
   473→                <label className={labelCls} htmlFor="qo-phone">
   474→                  {t("yourPhone")} *
   475→                </label>
   476→                <input
   477→                  id="qo-phone"
   478→                  required
   479→                  type="tel"
   480→                  inputMode="numeric"
   481→                  maxLength={10}
   482→                  value={form.phone}
   483→                  onChange={(e) => setForm((f) => ({ ...f, phone: e.target.value.replace(/\D/g, "").slice(0, 10) }))}
   484→                  className={cn(inputCls, "tracking-wider")}
   485→                  placeholder="0555 12 34 56"
   486→                  dir="ltr"
   487→                />
   488→              </div>
   489→            </div>
   490→
   491→            <DeliverySelector
   492→              value={delivery}
   493→              onChange={setDelivery}
   494→              onFee={(f, s) => {
   495→                setFee(f);
   496→                setFeeState(s);
   497→              }}
